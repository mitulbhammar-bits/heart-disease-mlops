"""Validated serving, readiness and low-cardinality request monitoring."""
import json
import logging
import os
import time
import uuid
from contextlib import asynccontextmanager
from pathlib import Path
from typing import Literal

import joblib
import numpy as np
import pandas as pd
from fastapi import FastAPI, Request, Response
from prometheus_client import CONTENT_TYPE_LATEST, Counter, Histogram, generate_latest
from pydantic import BaseModel, ConfigDict, Field

from heart_risk.settings import FEATURES, MODEL_PATH

logging.basicConfig(level=logging.INFO, format="%(message)s")
logger = logging.getLogger("heart_risk.requests")
REQUESTS = Counter("heart_api_requests_total", "Requests by route and status", ["route", "status"])
LATENCY = Histogram("heart_api_request_duration_seconds", "Request latency", ["route"])
PREDICTIONS = Counter("heart_api_predictions_total", "Predictions by class", ["prediction"])


class Patient(BaseModel):
    model_config = ConfigDict(extra="forbid", allow_inf_nan=False)
    age: float = Field(ge=1, le=120)
    sex: Literal[0, 1]
    chest_pain_type: Literal[1, 2, 3, 4]
    resting_blood_pressure: float = Field(gt=0, le=350)
    cholesterol: float = Field(gt=0, le=1000)
    high_fasting_blood_sugar: Literal[0, 1]
    resting_ecg: Literal[0, 1, 2]
    maximum_heart_rate: float = Field(gt=0, le=300)
    exercise_angina: Literal[0, 1]
    st_depression: float = Field(ge=0, le=10)
    st_slope: Literal[1, 2, 3]
    major_vessels: Literal[0, 1, 2, 3] | None = None
    thalassemia_test: Literal[3, 6, 7] | None = None


def patient_frame(patient: Patient) -> pd.DataFrame:
    return pd.DataFrame([patient.model_dump()], columns=FEATURES).replace({None: np.nan})


def create_app(model_path: Path | None = None) -> FastAPI:
    path = model_path or Path(os.getenv("MODEL_PATH", str(MODEL_PATH)))

    @asynccontextmanager
    async def lifespan(app):
        app.state.model = joblib.load(path)
        yield

    app = FastAPI(title="Heart Disease Learning API", version="1.0.0", lifespan=lifespan)

    @app.middleware("http")
    async def record_request(request: Request, call_next):
        start = time.perf_counter()
        request_id = uuid.uuid4().hex
        status = 500
        try:
            response = await call_next(request)
            status = response.status_code
            response.headers["X-Request-ID"] = request_id
            return response
        finally:
            route = request.scope.get("route")
            route_name = route.path if route else "unmatched"
            elapsed = time.perf_counter() - start
            REQUESTS.labels(route_name, str(status)).inc()
            LATENCY.labels(route_name).observe(elapsed)
            logger.info(json.dumps({"request_id": request_id, "method": request.method,
                                    "route": route_name, "status": status,
                                    "duration_ms": round(elapsed * 1000, 2)}))

    @app.get("/health")
    def health():
        return {"status": "ok"}

    @app.get("/ready")
    def ready():
        return {"status": "ready", "model_loaded": hasattr(app.state, "model")}

    @app.post("/predict")
    def predict(patient: Patient):
        probability = float(app.state.model.predict_proba(patient_frame(patient))[0, 1])
        prediction = int(probability >= 0.5)
        PREDICTIONS.labels(str(prediction)).inc()
        return {"prediction": prediction, "label": "present" if prediction else "absent",
                "heart_disease_probability": probability,
                "confidence": probability if prediction else 1 - probability,
                "threshold": 0.5, "model_version": "1.0.0"}

    @app.get("/metrics")
    def metrics():
        return Response(generate_latest(), media_type=CONTENT_TYPE_LATEST)

    return app


app = create_app()

"""One readable entry point for reproducible stages."""
import argparse
import json

from heart_risk.settings import ROOT


def main():
    parser = argparse.ArgumentParser(description="Heart disease learning pipeline")
    parser.add_argument("stage", choices=["download", "eda", "train", "predict", "all"])
    parser.add_argument("--refresh", action="store_true", help="Redownload official raw data")
    parser.add_argument("--input", type=str, default=str(ROOT / "examples/patient.json"))
    args = parser.parse_args()
    if args.stage in ("download", "all"):
        from heart_risk.data import acquire_data
        print(f"Prepared {len(acquire_data(args.refresh))} rows")
    if args.stage in ("eda", "all"):
        from heart_risk.exploration import create_eda
        create_eda()
    if args.stage in ("train", "all"):
        from heart_risk.training import train_models
        train_models()
    if args.stage == "predict":
        import joblib
        from pathlib import Path
        from heart_risk.api import Patient, patient_frame
        from heart_risk.settings import MODEL_PATH
        patient = Patient.model_validate(json.loads(Path(args.input).read_text()))
        probability = float(joblib.load(MODEL_PATH).predict_proba(patient_frame(patient))[0, 1])
        print(json.dumps({"prediction": int(probability >= 0.5), "probability": probability}, indent=2))


if __name__ == "__main__":
    main()

"""Download official UCI data; preserve missing values for fold-local imputation."""
import hashlib
import io
import json
import urllib.request
import zipfile

import pandas as pd

from heart_risk.settings import DATA_PATH, FRIENDLY_COLUMNS, RAW_COLUMNS, ROOT, SOURCE_URL


def clean_data(raw: pd.DataFrame) -> pd.DataFrame:
    if list(raw.columns) != RAW_COLUMNS:
        raise ValueError("Expected the 14 original Cleveland columns in source order")
    clean = raw.replace("?", float("nan")).apply(pd.to_numeric, errors="raise")
    if clean["diagnosis"].isna().any() or not clean["diagnosis"].isin(range(5)).all():
        raise ValueError("Diagnosis must contain UCI codes 0 through 4")
    clean["diagnosis"] = (clean["diagnosis"] > 0).astype(int)
    clean.columns = FRIENDLY_COLUMNS
    return clean.drop_duplicates().reset_index(drop=True)


def acquire_data(refresh: bool = False) -> pd.DataFrame:
    raw_path = ROOT / "data/raw/processed.cleveland.data"
    manifest_path = ROOT / "data/source_manifest.json"
    raw_path.parent.mkdir(parents=True, exist_ok=True)
    if refresh or not raw_path.exists():
        with urllib.request.urlopen(SOURCE_URL, timeout=60) as response:
            archive = zipfile.ZipFile(io.BytesIO(response.read()))
        member = next(n for n in archive.namelist() if n.endswith("processed.cleveland.data"))
        raw_path.write_bytes(archive.read(member))
    digest = hashlib.sha256(raw_path.read_bytes()).hexdigest()
    if manifest_path.exists() and not refresh:
        expected = json.loads(manifest_path.read_text())["source_sha256"]
        if expected != digest:
            raise ValueError("Raw source checksum mismatch; restore data or deliberately --refresh")
    raw = pd.read_csv(raw_path, names=RAW_COLUMNS, na_values="?")
    clean = clean_data(raw)
    DATA_PATH.parent.mkdir(parents=True, exist_ok=True)
    clean.to_csv(DATA_PATH, index=False)
    manifest = {"source_url": SOURCE_URL, "source_file": raw_path.name,
                "source_sha256": digest, "cleaned_sha256": hashlib.sha256(DATA_PATH.read_bytes()).hexdigest(),
                "source_rows": len(raw), "cleaned_rows": len(clean),
                "missing_values": clean.isna().sum().to_dict(),
                "target_rule": "diagnosis > 0 maps to 1; diagnosis == 0 maps to 0",
                "license": "CC BY 4.0", "citation": "https://doi.org/10.24432/C52P4X"}
    manifest_path.write_text(json.dumps(manifest, indent=2))
    return clean

"""Every fitted preprocessing operation lives inside the estimator pipeline."""
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler

from heart_risk.settings import CATEGORICAL_FEATURES, NUMERIC_FEATURES


def build_preprocessor() -> ColumnTransformer:
    continuous = Pipeline([("fill_missing", SimpleImputer(strategy="median")),
                           ("scale", StandardScaler())])
    categories = Pipeline([("fill_missing", SimpleImputer(strategy="most_frequent")),
                           ("encode", OneHotEncoder(handle_unknown="ignore", sparse_output=False))])
    return ColumnTransformer([("continuous", continuous, NUMERIC_FEATURES),
                              ("categories", categories, CATEGORICAL_FEATURES)])

from sklearn.ensemble import RandomForestClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline

from heart_risk.features import build_preprocessor
from heart_risk.settings import SEED


def build_model(name: str, **parameters) -> Pipeline:
    if name == "logistic_regression":
        classifier = LogisticRegression(max_iter=2000, random_state=SEED, **parameters)
    elif name == "random_forest":
        classifier = RandomForestClassifier(n_estimators=200, random_state=SEED,
                                            n_jobs=1, **parameters)
    else:
        raise ValueError(f"Unknown model: {name}")
    return Pipeline([("prepare_features", build_preprocessor()), ("classifier", classifier)])


def candidate_models():
    # Six small, intentional experiments keep the search understandable and inexpensive.
    for strength in (0.1, 1.0, 10.0):
        yield "logistic_regression", {"C": strength}
    for depth in (3, 6, None):
        yield "random_forest", {"max_depth": depth, "min_samples_leaf": 2}

==> heart_risk/training.py <==
"""Select with training-fold ROC-AUC; evaluate the winner once on held-out data."""
import hashlib
import json
import platform

import joblib
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import mlflow
import mlflow.sklearn
import pandas as pd
from sklearn.metrics import (ConfusionMatrixDisplay, RocCurveDisplay, accuracy_score,
                             precision_score, recall_score, roc_auc_score)
from sklearn.model_selection import StratifiedKFold, cross_validate, train_test_split

from heart_risk.data import acquire_data
from heart_risk.models import build_model, candidate_models
from heart_risk.settings import DATA_PATH, FEATURES, MODEL_PATH, ROOT, SEED

SCORING = {"accuracy": "accuracy", "precision": "precision", "recall": "recall",
           "roc_auc": "roc_auc"}


def train_models():
    data = acquire_data()
    features, target = data[FEATURES], data.heart_disease
    x_train, x_test, y_train, y_test = train_test_split(
        features, target, test_size=0.2, stratify=target, random_state=SEED)
    folds = StratifiedKFold(n_splits=5, shuffle=True, random_state=SEED)
    mlflow.set_tracking_uri(f"sqlite:///{ROOT / 'mlflow.db'}")
    experiment = mlflow.get_experiment_by_name("heart-risk-model-comparison")
    if experiment is None:
        mlflow.create_experiment("heart-risk-model-comparison",
                                 artifact_location=(ROOT / "mlartifacts").as_uri())
    mlflow.set_experiment("heart-risk-model-comparison")
    results, best = [], None
    figure_folder = ROOT / "reports/figures"
    figure_folder.mkdir(parents=True, exist_ok=True)
    for name, parameters in candidate_models():
        model = build_model(name, **parameters)
        scores = cross_validate(model, x_train, y_train, cv=folds, scoring=SCORING,
                                error_score="raise", n_jobs=1)
        metrics = {f"cv_{metric}_mean": float(scores[f"test_{metric}"].mean())
                   for metric in SCORING}
        metrics.update({f"cv_{metric}_std": float(scores[f"test_{metric}"].std())
                        for metric in SCORING})
        model.fit(x_train, y_train)
        with mlflow.start_run(run_name=f"{name}-{len(results)+1}") as run:
            mlflow.log_params({"model": name, "random_seed": SEED, "folds": 5,
                               "training_rows": len(x_train), "holdout_rows": len(x_test),
                               "threshold": 0.5, **parameters})
            mlflow.log_metrics(metrics)
            mlflow.set_tags({"purpose": "cv_candidate", "python": platform.python_version(),
                             "data_sha256": hashlib.sha256(DATA_PATH.read_bytes()).hexdigest()})
            fold_data = {metric: scores[f"test_{metric}"].tolist() for metric in SCORING}
            mlflow.log_dict(fold_data, "fold_metrics.json")
            mlflow.log_artifact(str(ROOT / "data/source_manifest.json"))
            fig, axis = plt.subplots(figsize=(7, 4))
            for metric, values in fold_data.items():
                axis.plot(range(1, 6), values, marker="o", label=metric)
            axis.set(xlabel="Validation fold", ylabel="Score", ylim=(0, 1.05),
                     title=f"{name}: {parameters}")
            axis.legend()
            fig.tight_layout()
            mlflow.log_figure(fig, "plots/cross_validation.png")
            plt.close(fig)
            mlflow.sklearn.log_model(model, name="pipeline",
                                     input_example=x_train.head(2))
            row = {"model": name, "parameters": parameters, "run_id": run.info.run_id, **metrics}
            results.append(row)
        if best is None or metrics["cv_roc_auc_mean"] > best["cv_roc_auc_mean"]:
            best = row
    # No candidate sees the holdout during selection. Keep the winner fitted on training data.
    winner = build_model(best["model"], **best["parameters"]).fit(x_train, y_train)
    probabilities = winner.predict_proba(x_test)[:, 1]
    predicted = (probabilities >= 0.5).astype(int)
    holdout = {"accuracy": accuracy_score(y_test, predicted),
               "precision": precision_score(y_test, predicted, zero_division=0),
               "recall": recall_score(y_test, predicted, zero_division=0),
               "roc_auc": roc_auc_score(y_test, probabilities)}
    for name, display in [("holdout_roc_curve", RocCurveDisplay.from_predictions),
                          ("holdout_confusion_matrix", ConfusionMatrixDisplay.from_predictions)]:
        display(y_test, probabilities if name.endswith("roc_curve") else predicted)
        plt.title(name.replace("_", " ").title())
        plt.tight_layout()
        plt.savefig(figure_folder / f"{name}.png", dpi=160)
        plt.close()
    MODEL_PATH.parent.mkdir(parents=True, exist_ok=True)
    joblib.dump(winner, MODEL_PATH)
    metadata = {"selected_model": best["model"], "selected_parameters": best["parameters"],
                "selection_metric": "mean 5-fold training-only ROC-AUC", "selected_run_id": best["run_id"],
                "cv_roc_auc": best["cv_roc_auc_mean"], "holdout_metrics": holdout,
                "threshold": 0.5, "seed": SEED, "train_rows": len(x_train), "test_rows": len(x_test),
                "features": FEATURES, "model_sha256": hashlib.sha256(MODEL_PATH.read_bytes()).hexdigest(),
                "data_sha256": hashlib.sha256(DATA_PATH.read_bytes()).hexdigest(),
                "python": platform.python_version(), "use": "Educational demonstration only"}
    (ROOT / "models/model_metadata.json").write_text(json.dumps(metadata, indent=2))
    pd.DataFrame(results).to_csv(ROOT / "reports/model_comparison.csv", index=False)
    pd.DataFrame({"source_row": x_test.index, "actual": y_test.values,
                  "probability": probabilities, "predicted": predicted}).to_csv(
                      ROOT / "reports/holdout_predictions.csv", index=False)
    (ROOT / "reports/split_indices.json").write_text(json.dumps(
        {"train": x_train.index.tolist(), "holdout": x_test.index.tolist()}, indent=2))
    with mlflow.start_run(run_name="selected-model-holdout"):
        mlflow.log_params({"selected_model": best["model"], "threshold": 0.5})
        mlflow.log_metrics({f"holdout_{key}": value for key, value in holdout.items()})
        mlflow.set_tag("purpose", "final_holdout_evaluation")
        mlflow.log_artifacts(str(figure_folder), artifact_path="plots")
        mlflow.log_artifact(str(ROOT / "reports/model_comparison.csv"))
        mlflow.log_artifact(str(ROOT / "models/model_metadata.json"))
        mlflow.log_artifact(str(ROOT / "reports/split_indices.json"))
        mlflow.sklearn.log_model(winner, name="selected_pipeline", input_example=x_train.head(2))
    fig, axis = plt.subplots(figsize=(9, 4))
    labels = [f"{row['model'].replace('_', ' ')} {i+1}" for i, row in enumerate(results)]
    axis.barh(labels, [r["cv_roc_auc_mean"] for r in results],
              xerr=[r["cv_roc_auc_std"] for r in results], color="#246a82")
    axis.set(xlabel="Mean CV ROC-AUC (+/- fold standard deviation)", xlim=(0, 1))
    fig.tight_layout()
    fig.savefig(figure_folder / "model_comparison.png", dpi=160)
    plt.close(fig)
    print(json.dumps(metadata, indent=2))
    return metadata

import joblib
import numpy as np
import pytest

from heart_risk.models import build_model
from heart_risk.settings import FEATURES


@pytest.mark.parametrize("name", ["logistic_regression", "random_forest"])
def test_model_handles_missing_and_unseen_categories(dataset, name):
    model = build_model(name).fit(dataset[FEATURES], dataset.heart_disease)
    example = dataset[FEATURES].head(2).copy()
    example.loc[example.index[0], "major_vessels"] = np.nan
    example.loc[example.index[1], "chest_pain_type"] = 99
    probabilities = model.predict_proba(example)
    assert probabilities.shape == (2, 2)
    assert np.isfinite(probabilities).all()
    assert np.allclose(probabilities.sum(axis=1), 1)


def test_serialization_preserves_predictions(tmp_path, fitted_model, dataset):
    path = tmp_path / "model.joblib"
    joblib.dump(fitted_model, path)
    loaded = joblib.load(path)
    example = dataset[FEATURES].head(4)
    np.testing.assert_allclose(loaded.predict_proba(example), fitted_model.predict_proba(example))


def test_imputer_learns_only_training_values(dataset):
    training = dataset[FEATURES].iloc[:50]
    model = build_model("logistic_regression").fit(training, dataset.heart_disease.iloc[:50])
    numeric = model.named_steps["prepare_features"].named_transformers_["continuous"]
    assert numeric.named_steps["fill_missing"].statistics_[0] == training.age.median()

    with pytest.raises(ValueError, match="Unknown"):
        build_model("unsupported")
